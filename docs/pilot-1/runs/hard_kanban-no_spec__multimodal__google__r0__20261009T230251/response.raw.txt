import { Component, ElementRef, ViewChild } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { NgIcon, provideIcons } from '@ng-icons/core';
import { lucidePencil, lucideTrash2, lucidePlus, lucideCheck, lucideX } from '@ng-icons/lucide';

export interface KanbanCard {
  id: string;
  description: string;
}

export interface KanbanColumn {
  id: string;
  title: string;
  headerColor: string;
  bgColor: string;
  cards: KanbanCard[];
}

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, FormsModule, NgIcon],
  viewProviders: [
    provideIcons({
      lucidePencil,
      lucideTrash2,
      lucidePlus,
      lucideCheck,
      lucideX
    })
  ],
  templateUrl: './app.html',
  styleUrls: ['./app.css']
})
export class App {
  @ViewChild('newCardInput') newCardInputRef?: ElementRef<HTMLTextAreaElement>;
  @ViewChild('editInput') editInputRef?: ElementRef<HTMLTextAreaElement>;

  // As 4 colunas solicitadas com as exatas cores da imagem
  columns: KanbanColumn[] = [
    {
      id: 'todo',
      title: 'Para fazer',
      headerColor: '#137ec2', // Azul forte
      bgColor: '#92b5ce',     // Azul tom fraco
      cards: [
        {
          id: 'card-1',
          description: 'Testes end-to-end para garantir o funcionamento do site'
        },
        {
          id: 'card-2',
          description: 'Testes unitários no backend para garantir respostas consistentes'
        }
      ]
    },
    {
      id: 'in-progress',
      title: 'Em Progresso',
      headerColor: '#c9800e', // Laranja / Ocre forte
      bgColor: '#cbb399',     // Ocre tom fraco
      cards: [
        {
          id: 'card-3',
          description: 'Testes end-to-end para garantir o funcionamento do site'
        },
        {
          id: 'card-4',
          description: 'Testes unitários no backend para garantir respostas consistentes'
        }
      ]
    },
    {
      id: 'done',
      title: 'Feito',
      headerColor: '#4e12c4', // Roxo forte
      bgColor: '#9c94c9',     // Roxo tom fraco
      cards: [
        {
          id: 'card-5',
          description: 'Testes end-to-end para garantir o funcionamento do site'
        },
        {
          id: 'card-6',
          description: 'Testes unitários no backend para garantir respostas consistentes'
        }
      ]
    },
    {
      id: 'approved',
      title: 'Aprovado',
      headerColor: '#1b671a', // Verde forte
      bgColor: '#c1cc9e',     // Verde tom fraco
      cards: [
        {
          id: 'card-7',
          description: 'Testes end-to-end para garantir o funcionamento do site'
        },
        {
          id: 'card-8',
          description: 'Testes unitários no backend para garantir respostas consistentes'
        }
      ]
    }
  ];

  // Estado da adição de novo card na 1ª coluna
  isAddingCard = false;
  newCardText = '';

  // Estado de edição de card existente
  editingCardId: string | null = null;
  editingText = '';

  // Estado do Drag and Drop
  draggedCard: KanbanCard | null = null;
  draggedCardColIndex: number = -1;
  draggedCardIndex: number = -1;

  dragOverColumnIndex: number | null = null;
  dragOverCardId: string | null = null;
  dropPosition: 'before' | 'after' | null = null;

  // Regra de transição do Kanban:
  // - Apenas 1 coluna à direita (ex: 0 -> 1, 1 -> 2, 2 -> 3)
  // - Quantas colunas quiser para a esquerda (ex: 3 -> 0, 3 -> 1, 2 -> 0, etc.)
  // - Mesma coluna (reordenação) é permitida
  isDropAllowed(fromColIndex: number, toColIndex: number): boolean {
    if (fromColIndex < 0 || toColIndex < 0) return false;
    if (fromColIndex === toColIndex) return true; // Reordenação na mesma coluna
    if (toColIndex < fromColIndex) return true;   // Para a esquerda: quantos passos quiser
    if (toColIndex === fromColIndex + 1) return true; // Para a direita: APENAS 1 etapa
    return false; // Pular mais de 1 etapa à direita é proibido
  }

  // --- Handlers de Adição de Cartão ---
  openAddCard(): void {
    this.isAddingCard = true;
    this.newCardText = '';
    setTimeout(() => {
      this.newCardInputRef?.nativeElement.focus();
    }, 50);
  }

  confirmAddCard(): void {
    const trimmed = this.newCardText.trim();
    if (trimmed) {
      this.columns[0].cards.push({
        id: 'card-' + Date.now(),
        description: trimmed
      });
    }
    this.cancelAddCard();
  }

  cancelAddCard(): void {
    this.isAddingCard = false;
    this.newCardText = '';
  }

  // --- Handlers de Edição ---
  startEditing(card: KanbanCard): void {
    this.editingCardId = card.id;
    this.editingText = card.description;
    setTimeout(() => {
      this.editInputRef?.nativeElement.focus();
    }, 50);
  }

  saveEditingCard(card: KanbanCard): void {
    const trimmed = this.editingText.trim();
    if (trimmed) {
      card.description = trimmed;
    }
    this.editingCardId = null;
    this.editingText = '';
  }

  cancelEditing(): void {
    this.editingCardId = null;
    this.editingText = '';
  }

  // --- Handlers de Exclusão ---
  deleteCard(colIndex: number, cardId: string): void {
    this.columns[colIndex].cards = this.columns[colIndex].cards.filter(c => c.id !== cardId);
  }

  // --- Drag and Drop Handlers ---
  onDragStart(event: DragEvent, card: KanbanCard, colIndex: number, cardIndex: number): void {
    this.draggedCard = card;
    this.draggedCardColIndex = colIndex;
    this.draggedCardIndex = cardIndex;

    if (event.dataTransfer) {
      event.dataTransfer.effectAllowed = 'move';
      event.dataTransfer.setData('text/plain', card.id);
    }
  }

  onDragEnd(): void {
    this.draggedCard = null;
    this.draggedCardColIndex = -1;
    this.draggedCardIndex = -1;
    this.dragOverColumnIndex = null;
    this.dragOverCardId = null;
    this.dropPosition = null;
  }

  onDragOverColumn(event: DragEvent, colIndex: number): void {
    if (!this.draggedCard) return;

    if (this.isDropAllowed(this.draggedCardColIndex, colIndex)) {
      event.preventDefault();
      if (event.dataTransfer) {
        event.dataTransfer.dropEffect = 'move';
      }
      this.dragOverColumnIndex = colIndex;
    } else {
      if (event.dataTransfer) {
        event.dataTransfer.dropEffect = 'none';
      }
    }
  }

  onDragLeaveColumn(event: DragEvent, colIndex: number): void {
    if (this.dragOverColumnIndex === colIndex) {
      this.dragOverColumnIndex = null;
    }
  }

  onDropColumn(event: DragEvent, targetColIndex: number): void {
    event.preventDefault();
    if (!this.draggedCard || !this.isDropAllowed(this.draggedCardColIndex, targetColIndex)) {
      this.onDragEnd();
      return;
    }

    const sourceColIndex = this.draggedCardColIndex;
    const card = this.draggedCard;

    // Se o drop não foi em cima de um card específico, adiciona no final da coluna
    if (!this.dragOverCardId) {
      // Remove da coluna original
      this.columns[sourceColIndex].cards = this.columns[sourceColIndex].cards.filter(c => c.id !== card.id);
      // Adiciona na coluna destino
      this.columns[targetColIndex].cards.push(card);
    }

    this.onDragEnd();
  }

  onDragOverCard(event: DragEvent, colIndex: number, cardIndex: number): void {
    if (!this.draggedCard) return;

    if (this.isDropAllowed(this.draggedCardColIndex, colIndex)) {
      event.preventDefault();
      event.stopPropagation();

      const targetCard = this.columns[colIndex].cards[cardIndex];
      this.dragOverCardId = targetCard.id;
      this.dragOverColumnIndex = colIndex;

      const targetEl = event.currentTarget as HTMLElement;
      const rect = targetEl.getBoundingClientRect();
      const relativeY = event.clientY - rect.top;

      this.dropPosition = relativeY > rect.height / 2 ? 'after' : 'before';
    }
  }

  onDragLeaveCard(): void {
    this.dragOverCardId = null;
    this.dropPosition = null;
  }

  onDropCard(event: DragEvent, targetColIndex: number, targetCardIndex: number): void {
    event.preventDefault();
    event.stopPropagation();

    if (!this.draggedCard || !this.isDropAllowed(this.draggedCardColIndex, targetColIndex)) {
      this.onDragEnd();
      return;
    }

    const sourceColIndex = this.draggedCardColIndex;
    const card = this.draggedCard;
    const currentCardList = this.columns[sourceColIndex].cards;
    const fromIndex = currentCardList.findIndex(c => c.id === card.id);

    // Remove do local original
    if (fromIndex !== -1) {
      currentCardList.splice(fromIndex, 1);
    }

    // Calcula índice de destino
    const targetCardList = this.columns[targetColIndex].cards;
    let insertIndex = targetCardIndex;

    // Se movendo na mesma coluna e o item original estava antes do alvo
    if (sourceColIndex === targetColIndex && fromIndex < targetCardIndex) {
      insertIndex = targetCardIndex;
    }

    if (this.dropPosition === 'after') {
      insertIndex++;
    }

    // Garante limites válidos
    insertIndex = Math.max(0, Math.min(insertIndex, targetCardList.length));
    targetCardList.splice(insertIndex, 0, card);

    this.onDragEnd();
  }
}
