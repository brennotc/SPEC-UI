export class App {
  readonly columns: ColumnConfig[] = [
    {
      id: 'para-fazer',
      title: 'Para fazer',
      headerColorHex: '#0B73B7',
      bodyColorHex: '#9FBACC'
    },
    {

      title: 'Em Progresso',

    {

      title: 'Feito',

    },
    {
      id: 'aprovado',
      title: 'Aprovado',
      headerColorHex: '#105610',
      bodyColorHex: '#C5CC9F'
    }
  ];

  // Estado global dos cards em memória
  readonly cards = signal<KanbanCard[]>([
    // Coluna Para Fazer
    {
      id: 'card-1',
      description: 'Testes end-to-end para garantir o funcionamento do site',
      columnId: 'para-fazer',

    },

      id: 'card-2',
      description: 'Testes unitários no backend para garantir respostas consistentes',
      columnId: 'para-fazer',
      createdAt: new Date().toISOString()

    // Coluna Em Progresso
    {
      id: 'card-3',
      description: 'Testes end-to-end para garantir o funcionamento do site',
      columnId: 'em-progresso',
      createdAt: new Date().toISOString()
    },
    {
      id: 'card-4',
      description: 'Testes unitários no backend para garantir respostas consistentes',
      columnId: 'em-progresso',
      createdAt: new Date().toISOString()
    },
    // Coluna Feito
    {
      id: 'card-5',
      description: 'Testes end-to-end para garantir o funcionamento do site',
      columnId: 'feito',
      createdAt: new Date().toISOString()
    },
    {
      id: 'card-6',
      description: 'Testes unitários no backend para garantir respostas consistentes',
      columnId: 'feito',
      createdAt: new Date().toISOString()
    },
    // Coluna Aprovado
    {
      id: 'card-7',
      description: 'Testes end-to-end para garantir o funcionamento do site',
      columnId: 'aprovado',
      createdAt: new Date().toISOString()
    },
    {
      id: 'card-8',
      description: 'Testes unitários no backend para garantir respostas consistentes',
      columnId: 'aprovado',
      createdAt: new Date().toISOString()
    }
  ]);

  // Estado de criação de novo card
  readonly isCreating = signal<boolean>(false);
  newCardText = '';

  // Estado de edição de card existente
  readonly editingCardId = signal<string | null>(null);
  editingText = '';

  // Estado de Drag & Drop
  readonly draggedCard = signal<KanbanCard | null>(null);
  readonly draggedColIndex = signal<number | null>(null);
  readonly draggedCardIndex = signal<number | null>(null);
  readonly dragOverColIndex = signal<number | null>(null);

  isDropAllowed = computed(() => {
    const fromCol = this.draggedColIndex();
    const toCol = this.dragOverColIndex();
    if (fromCol === null || toCol === null) return false;
    return this.checkDropRule(fromCol, toCol);
  });

  getCardsByColumn(columnId: ColumnId): KanbanCard[] {
    return this.cards().filter((c) => c.columnId === columnId);
  }

  // Regra de movimentação estrita:
  // - Mesma coluna: permitido (reordenação)
  // - Para esquerda: permitido qualquer número de colunas (recuo N)
  // - Para direita: permitido APENAS coluna imediatamente seguinte (+1)
  checkDropRule(fromCol: number, toCol: number): boolean {
    if (fromCol === toCol) return true;
    if (toCol < fromCol) return true;
    if (toCol === fromCol + 1) return true;
    return false;
  }

  // Fluxo de Criação
  startCreating() {
    this.newCardText = '';
    this.isCreating.set(true);
  }

  confirmCreate() {
    const text = this.newCardText.trim();
    if (text) {
      const newCard: KanbanCard = {
        id: 'card-' + Date.now(),
        description: text,
        columnId: 'para-fazer',
        createdAt: new Date().toISOString()
      };
      this.cards.update((current) => [...current, newCard]);
    }
    this.newCardText = '';
    this.isCreating.set(false);
  }

  cancelCreate() {
    this.newCardText = '';
    this.isCreating.set(false);
  }

  // Fluxo de Edição
  startEditing(card: KanbanCard) {
    this.editingCardId.set(card.id);
    this.editingText = card.description;
  }

  saveEditing(cardId: string) {
    const text = this.editingText.trim();
    if (text) {
      this.cards.update((cards) =>
        cards.map((c) => (c.id === cardId ? { ...c, description: text } : c))
      );
    }
    this.editingCardId.set(null);
    this.editingText = '';
  }

  cancelEditing() {
    this.editingCardId.set(null);
    this.editingText = '';
  }

  // Fluxo de Exclusão
  deleteCard(cardId: string) {
    this.cards.update((cards) => cards.filter((c) => c.id !== cardId));
  }

  // Eventos de Drag & Drop
  onCardDragStart(
    event: DragEvent,
    card: KanbanCard,
    colIndex: number,
    cardIndex: number
  ) {
    if (this.editingCardId() !== null || this.isCreating()) {
      event.preventDefault();
      return;
    }
    this.draggedCard.set(card);
    this.draggedColIndex.set(colIndex);
    this.draggedCardIndex.set(cardIndex);

    if (event.dataTransfer) {
      event.dataTransfer.effectAllowed = 'move';
      event.dataTransfer.setData('text/plain', card.id);
    }
  }

  onCardDragEnd() {
    this.draggedCard.set(null);
    this.draggedColIndex.set(null);
    this.draggedCardIndex.set(null);
    this.dragOverColIndex.set(null);
  }

  onColumnDragOver(event: DragEvent, colIndex: number) {
    const fromCol = this.draggedColIndex();
    if (fromCol === null) return;

    if (this.checkDropRule(fromCol, colIndex)) {
      event.preventDefault();
      if (event.dataTransfer) {
        event.dataTransfer.dropEffect = 'move';
      }
      this.dragOverColIndex.set(colIndex);
    } else {
      if (event.dataTransfer) {
        event.dataTransfer.dropEffect = 'none';
      }
      this.dragOverColIndex.set(null);
    }
  }

  onColumnDragLeave(event: DragEvent, colIndex: number) {
    if (this.dragOverColIndex() === colIndex) {
      this.dragOverColIndex.set(null);
    }
  }

  onCardDragOver(event: DragEvent, colIndex: number, _cardIndex: number) {
    this.onColumnDragOver(event, colIndex);
  }

  onColumnDrop(event: DragEvent, targetColIndex: number) {
    event.preventDefault();
    this.executeDrop(targetColIndex);
  }

  onCardDrop(event: DragEvent, targetColIndex: number, targetCardIndex: number) {
    event.preventDefault();
    event.stopPropagation();
    this.executeDrop(targetColIndex, targetCardIndex);
  }

  private executeDrop(targetColIndex: number, targetCardIndex?: number) {
    const card = this.draggedCard();
    const fromColIndex = this.draggedColIndex();

    if (!card || fromColIndex === null) {
      this.onCardDragEnd();
      return;
    }

    // Validação estrita da regra:
    if (!this.checkDropRule(fromColIndex, targetColIndex)) {
      // Movimentação proibida: snap back
      this.onCardDragEnd();
      return;
    }

    const targetColumn = this.columns[targetColIndex];
    const currentList = [...this.cards()];
    const oldIndex = currentList.findIndex((c) => c.id === card.id);

    if (oldIndex === -1) {
      this.onCardDragEnd();
      return;
    }

    // Remove da posição original
    const [removedCard] = currentList.splice(oldIndex, 1);
    const updatedCard: KanbanCard = {
      ...removedCard,
      columnId: targetColumn.id
    };

    if (targetCardIndex !== undefined && targetCardIndex >= 0) {
      // Inserir antes ou depois do card alvo na coluna destino
      const colCards = currentList.filter((c) => c.columnId === targetColumn.id);
      if (targetCardIndex < colCards.length) {
        const refCard = colCards[targetCardIndex];
        const insertPosition = currentList.indexOf(refCard);
        currentList.splice(insertPosition, 0, updatedCard);
      } else {
        currentList.push(updatedCard);
      }
    } else {
      // Inserir no final da lista da coluna de destino
      currentList.push(updatedCard);
    }

    this.cards.set(currentList);
    this.onCardDragEnd();
  }
}
