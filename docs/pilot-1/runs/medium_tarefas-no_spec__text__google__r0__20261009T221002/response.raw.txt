export class App {
  // Dados das tarefas
  tasks = signal<TaskItem[]>([]);

  // Filtros
  filterDate = signal<string>('');
  filterPriority = signal<string>('TODAS');

  // Nova Tarefa Form
  newDate = signal<string>('');
  newPriority = signal<PriorityType | ''>('MEDIO');
  newTitle = signal<string>('');

  // Edição inline
  editingTaskId = signal<string | null>(null);
  editingText = signal<string>('');

  constructor() {
    this.initInitialTasks();
  }

  // Gera datas de referência (Hoje, Amanhã, Depois de amanhã)
  private getFormattedDateOffset(offsetDays: number): string {
    const d = new Date();
    d.setDate(d.getDate() + offsetDays);
    const day = String(d.getDate()).padStart(2, '0');
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const year = d.getFullYear();
    return `${day}/${month}/${year}`;
  }

  private initInitialTasks() {
    const today = this.getFormattedDateOffset(0);
    const tomorrow = this.getFormattedDateOffset(1);
    const dayAfter = this.getFormattedDateOffset(2);

    this.newDate.set(today);

    this.tasks.set([
      {
        id: '1',
        title: 'Enviar relatório financeiro trimestral para a diretoria',
        date: today,
        priority: 'URGENTE',
        completed: false
      },
      {
        id: '2',
        title: 'Reunião de alinhamento com equipe de engenharia',
        date: today,
        priority: 'MEDIO',
        completed: true
      },
      {
        id: '3',
        title: 'Assistir webinar de novas tecnologias web',
        date: today,
        priority: 'RELAXA',
        completed: false
      },
      {
        id: '4',
        title: 'Finalizar testes de integração do módulo de pagamento',
        date: tomorrow,
        priority: 'URGENTE',
        completed: false
      },
      {
        id: '5',
        title: 'Organizar documentação interna e repositórios',
        date: tomorrow,
        priority: 'MEDIO',
        completed: false
      },
      {
        id: '6',
        title: 'Pesquisar inspirações de design e referências visuais',
        date: dayAfter,
        priority: 'RELAXA',
        completed: false
      }
    ]);
  }

  completedCount = computed(() => {
    return this.tasks().filter(t => t.completed).length;
  });

  isFormValid = computed(() => {
    return (
      this.newTitle().trim().length > 0 &&
      this.newDate().trim().length === 10 &&
      this.newPriority() !== ''
    );
  });

  // Agrupamento ordenado por dias: Hoje, Amanhã, depois os dias escritos
  groupedTasks = computed(() => {
    const todayStr = this.getFormattedDateOffset(0);
    const tomorrowStr = this.getFormattedDateOffset(1);

    const fDate = this.filterDate().trim();
    const fPriority = this.filterPriority();

    // Filtra tarefas
    const filtered = this.tasks().filter(t => {
      if (fDate && t.date !== fDate) return false;
      if (fPriority !== 'TODAS' && t.priority !== fPriority) return false;
      return true;
    });

    // Mapeia por chave de data
    const map = new Map<string, TaskItem[]>();
    for (const t of filtered) {
      if (!map.has(t.date)) {
        map.set(t.date, []);
      }
      map.get(t.date)!.push(t);
    }

    const groups: GroupedTasks[] = [];

    map.forEach((taskList, dateStr) => {
      const parsedTimestamp = this.parseDateToTimestamp(dateStr);
      let label = '';

      if (dateStr === todayStr) {
        label = 'Hoje';
      } else if (dateStr === tomorrowStr) {
        label = 'Amanhã';
      } else {
        label = this.getWrittenDayName(dateStr);
      }

      groups.push({
        groupLabel: label,
        dateStr,
        timestamp: parsedTimestamp,
        tasks: taskList
      });
    });

    // Ordenação: Hoje sempre primeiro, Amanhã segundo, depois cronológico
    groups.sort((a, b) => {
      if (a.dateStr === todayStr) return -1;
      if (b.dateStr === todayStr) return 1;
      if (a.dateStr === tomorrowStr) return -1;
      if (b.dateStr === tomorrowStr) return 1;
      return a.timestamp - b.timestamp;
    });

    return groups;
  });

  private parseDateToTimestamp(dateStr: string): number {
    const parts = dateStr.split('/');
    if (parts.length === 3) {
      const day = parseInt(parts[0], 10);
      const month = parseInt(parts[1], 10) - 1;
      const year = parseInt(parts[2], 10);
      const d = new Date(year, month, day);
      return d.getTime() || 0;
    }
    return 0;
  }

  private getWrittenDayName(dateStr: string): string {
    const parts = dateStr.split('/');
    if (parts.length === 3) {
      const day = parseInt(parts[0], 10);
      const month = parseInt(parts[1], 10) - 1;
      const year = parseInt(parts[2], 10);
      const date = new Date(year, month, day);

      const daysOfWeek = [
        'Domingo',
        'Segunda-feira',
        'Terça-feira',
        'Quarta-feira',
        'Quinta-feira',
        'Sexta-feira',
        'Sábado'
      ];
      const dayName = daysOfWeek[date.getDay()];
      if (dayName) {
        return dayName;
      }
    }
    return dateStr;
  }

  // Máscara de data DD/MM/AAAA
  private applyMask(value: string): string {
    const digits = value.replace(/\D/g, '').slice(0, 8);
    if (digits.length > 4) {
      return `${digits.slice(0, 2)}/${digits.slice(2, 4)}/${digits.slice(4)}`;
    }
    if (digits.length > 2) {
      return `${digits.slice(0, 2)}/${digits.slice(2)}`;
    }
    return digits;
  }

  onFilterDateInput(event: Event) {
    const input = event.target as HTMLInputElement;
    const masked = this.applyMask(input.value);
    input.value = masked;
    this.filterDate.set(masked);
  }

  onNewDateInput(event: Event) {
    const input = event.target as HTMLInputElement;
    const masked = this.applyMask(input.value);
    input.value = masked;
    this.newDate.set(masked);
  }

  toggleTask(taskId: string) {
    this.tasks.update(all =>
      all.map(t => (t.id === taskId ? { ...t, completed: !t.completed } : t))
    );
  }

  deleteTask(taskId: string) {
    this.tasks.update(all => all.filter(t => t.id !== taskId));
    if (this.editingTaskId() === taskId) {
      this.editingTaskId.set(null);
    }
  }

  startEditing(task: TaskItem) {
    this.editingTaskId.set(task.id);
    this.editingText.set(task.title);
  }

  saveEditing(taskId: string) {
    const newTitle = this.editingText().trim();
    if (newTitle) {
      this.tasks.update(all =>
        all.map(t => (t.id === taskId ? { ...t, title: newTitle } : t))
      );
    }
    this.editingTaskId.set(null);
  }

  cancelEditing() {
    this.editingTaskId.set(null);
  }

  addTask(event: Event) {
    event.preventDefault();
    const title = this.newTitle().trim();
    const date = this.newDate().trim();
    const priority = this.newPriority() as PriorityType;

    if (!title || date.length !== 10 || !priority) {
      return;
    }

    const newTask: TaskItem = {
      id: Date.now().toString(),
      title,
      date,
      priority,
      completed: false
    };

    this.tasks.update(all => [...all, newTask]);

    // Limpa campos mantendo a data corrente para agilidade
    this.newTitle.set('');
  }
}
