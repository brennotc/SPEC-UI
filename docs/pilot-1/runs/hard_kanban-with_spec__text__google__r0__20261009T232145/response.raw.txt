```ts file=src/app/models/kanban.model.ts
export interface KanbanCard {

}

export interface NewCardState {

  tempDescription: string;
}
```

```ts file=src/app/app.ts
import { Component, signal, computed, HostListener } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { NgIconComponent, provideIcons } from '@ng-icons/core';
import { 
  lucideList, 
  lucidePencil, 
  lucideTrash2, 
  lucidePlus, 
  lucideCheck, 
  lucideX,
  lucideAlertCircle
} from '@ng-icons/lucide';
import { ColumnId, KanbanCard, ColumnConfig, NewCardState } from './models/kanban.model';

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, FormsModule, NgIconComponent],
  viewProviders: [
    provideIcons({
      lucideList,
      lucidePencil,
      lucideTrash2,
      lucidePlus,
      lucideCheck,
      lucideX,
      lucideAlertCircle
    })
  ],
  templateUrl: './app.html'
})
export class App {
  readonly columns: ColumnConfig[] = [
    {
      id: 'para-fazer',
      title: 'Para Fazer',
      headerColorClass: '#0B73B7',
      bodyColorClass: '#9FBACC'
    },
    {
      id: 'em-progresso',
      title: 'Em Progresso',
      headerColorClass: '#B77A0B',
      bodyColorClass: '#CCB59F'
    },
    {
      id: 'feito',
      title: 'Feito',
      headerColorClass: '#440BB7',
      bodyColorClass: '#AA9FCC'
    },
    {
      id: 'aprovado',
      title: 'Aprovado',
      headerColorClass: '#105610',
      bodyColorClass: '#C5CC9F'
    }
  ];

  // Estado dos Cards
  readonly cards = signal<KanbanCard[]>([
    {
      id: 'card-1',
      description: 'Estruturar protótipo UI',
      columnId: 'para-fazer',
      createdAt: new Date().toISOString()
    },
    {
      id: 'card-2',
      description: 'Implementar fluxo de Auth',
      columnId: 'em-progresso',
      createdAt: new Date().toISOString()
    },
    {
      id: 'card-3',
      description: 'Configurar banco de dados',
      columnId: 'feito',
      createdAt: new Date().toISOString()
    },
    {
      id: 'card-4',
      description: 'Homologação com cliente',
      columnId: 'aprovado',
      createdAt: new Date().toISOString()
    }
  ]);

  // Estado do formulário de novo card na coluna "Para Fazer"
  readonly newCardState = signal<NewCardState>({
    isCreating: false,
    tempDescription: ''
  });

  // Estado do card atualmente em edição inline
  readonly editingCardId = signal<string | null>(null);
  readonly editingCardText = signal<string>('');

  // Drag-and-drop state
  readonly draggedCardId = signal<string | null>(null);
  readonly dragOverColumnId = signal<ColumnId | null>(null);
  readonly dragOverCardId = signal<string | null>(null);
  readonly dropPosition = signal<'above' | 'below' | null>(null);

  // Notificação/Alerta efêmero para tentativas de movimentos ilegais
  readonly invalidMoveWarning = signal<string | null>(null);
  private warningTimeout: any = null;

  // Mapa de ordem de colunas para validação de movimento
  private readonly columnOrder: Record<ColumnId, number> = {
    'para-fazer': 0,
    'em-progresso': 1,
    'feito': 2,
    'aprovado': 3
  };

  getCardsForColumn(colId: ColumnId): KanbanCard[] {
    return this.cards().filter(card => card.columnId === colId);
  }

  // --- Criação de Novo Card ---
  startCreatingCard(): void {
    this.newCardState.set({
      isCreating: true,
      tempDescription: ''
    });
    setTimeout(() => {
      const input = document.getElementById('new-card-input') as HTMLInputElement;
      input?.focus();
    }, 50);
  }

  cancelCreatingCard(): void {
    this.newCardState.set({
      isCreating: false,
      tempDescription: ''
    });
  }

  confirmCreatingCard(): void {
    const desc = this.newCardState().tempDescription.trim();
    if (!desc) {
      this.cancelCreatingCard();
      return;
    }

    const newCard: KanbanCard = {
      id: `card-${Date.now()}`,
      description: desc,
      columnId: 'para-fazer',
      createdAt: new Date().toISOString()
    };

    // Insere como o último item da lista da coluna "Para Fazer"
    this.cards.update(list => [...list, newCard]);
    this.cancelCreatingCard();
  }

  // --- Edição de Card Existente ---
  startEditingCard(card: KanbanCard, event?: MouseEvent): void {
    event?.stopPropagation();
    this.editingCardId.set(card.id);
    this.editingCardText.set(card.description);
    setTimeout(() => {
      const input = document.getElementById(`edit-input-${card.id}`) as HTMLInputElement;
      input?.focus();
      input?.select();
    }, 50);
  }

  cancelEditingCard(event?: MouseEvent): void {
    event?.stopPropagation();
    this.editingCardId.set(null);
    this.editingCardText.set('');
  }

  confirmEditingCard(cardId: string, event?: MouseEvent): void {
    event?.stopPropagation();
    const updatedText = this.editingCardText().trim();
    if (updatedText) {
      this.cards.update(list =>
        list.map(c => (c.id === cardId ? { ...c, description: updatedText } : c))
      );
    }
    this.editingCardId.set(null);
    this.editingCardText.set('');
  }

  // --- Exclusão de Card ---
  deleteCard(cardId: string, event?: MouseEvent): void {
    event?.stopPropagation();
    this.cards.update(list => list.filter(c => c.id !== cardId));
    if (this.editingCardId() === cardId) {
      this.editingCardId.set(null);
    }
  }

  // --- Regras Estritas de Movimentação e Drag and Drop ---
  canMoveBetween(sourceCol: ColumnId, targetCol: ColumnId): boolean {
    if (sourceCol === targetCol) return true;
    const sourceIdx = this.columnOrder[sourceCol];
    const targetIdx = this.columnOrder[targetCol];

    // Para a direita: permitido APENAS 1 etapa à frente
    if (targetIdx > sourceIdx) {
      return targetIdx === sourceIdx + 1;
    }

    // Para a esquerda: permitido N etapas
    return targetIdx < sourceIdx;
  }

  onDragStart(event: DragEvent, card: KanbanCard): void {
    if (this.editingCardId() === card.id) {
      event.preventDefault();
      return;
    }
    this.draggedCardId.set(card.id);
    if (event.dataTransfer) {
      event.dataTransfer.setData('text/plain', card.id);
      event.dataTransfer.effectAllowed = 'move';
    }
  }

  onDragEnd(): void {
    this.draggedCardId.set(null);
    this.dragOverColumnId.set(null);
    this.dragOverCardId.set(null);
    this.dropPosition.set(null);
  }

  onDragOverColumn(event: DragEvent, columnId: ColumnId): void {
    event.preventDefault();
    if (event.dataTransfer) {
      event.dataTransfer.dropEffect = 'move';
    }
    if (this.dragOverColumnId() !== columnId) {
      this.dragOverColumnId.set(columnId);
    }
  }

  onDragLeaveColumn(event: DragEvent, columnId: ColumnId): void {
    const relatedTarget = event.relatedTarget as HTMLElement;
    const colElement = (event.currentTarget as HTMLElement);
    if (!colElement.contains(relatedTarget)) {
      if (this.dragOverColumnId() === columnId) {
        this.dragOverColumnId.set(null);
      }
    }
  }

  onDragOverCard(event: DragEvent, targetCard: KanbanCard): void {
    event.preventDefault();
    event.stopPropagation();
    const draggedId = this.draggedCardId();
    if (!draggedId || draggedId === targetCard.id) return;

    this.dragOverColumnId.set(targetCard.columnId);
    this.dragOverCardId.set(targetCard.id);

    const rect = (event.currentTarget as HTMLElement).getBoundingClientRect();
    const midY = rect.top + rect.height / 2;
    if (event.clientY < midY) {
      this.dropPosition.set('above');
    } else {
      this.dropPosition.set('below');
    }
  }

  onDropOnColumn(event: DragEvent, targetColumnId: ColumnId): void {
    event.preventDefault();
    event.stopPropagation();
    const draggedId = this.draggedCardId();
    if (!draggedId) {
      this.onDragEnd();
      return;
    }

    const currentCard = this.cards().find(c => c.id === draggedId);
    if (!currentCard) {
      this.onDragEnd();
      return;
    }

    const sourceColId = currentCard.columnId;

    // Se estiver mudando de coluna, validar regra estrita
    if (sourceColId !== targetColumnId) {
      if (!this.canMoveBetween(sourceColId, targetColumnId)) {
        this.triggerSnapBackNotification(
          `Avanço proibido: Só é permitido avançar 1 coluna por vez!`
        );
        this.onDragEnd();
        return;
      }

      // Mudar coluna e colocar no final
      this.cards.update(allCards => {
        const remaining = allCards.filter(c => c.id !== draggedId);
        const updatedCard = { ...currentCard, columnId: targetColumnId };
        return [...remaining, updatedCard];
      });
    }

    this.onDragEnd();
  }

  onDropOnCard(event: DragEvent, targetCard: KanbanCard): void {
    event.preventDefault();
    event.stopPropagation();
    const draggedId = this.draggedCardId();
    if (!draggedId || draggedId === targetCard.id) {
      this.onDragEnd();
      return;
    }

    const currentCard = this.cards().find(c => c.id === draggedId);
    if (!currentCard) {
      this.onDragEnd();
      return;
    }

    const sourceColId = currentCard.columnId;
    const targetColId = targetCard.columnId;

    // Se for entre colunas diferentes, checar regra
    if (sourceColId !== targetColId) {
      if (!this.canMoveBetween(sourceColId, targetColId)) {
        this.triggerSnapBackNotification(
          `Avanço proibido: Só é permitido avançar 1 coluna por vez!`
        );
        this.onDragEnd();
        return;
      }
    }

    const position = this.dropPosition() || 'below';

    this.cards.update(allCards => {
      const listWithoutDragged = allCards.filter(c => c.id !== draggedId);
      const targetIndex = listWithoutDragged.findIndex(c => c.id === targetCard.id);
      const updatedCard = { ...currentCard, columnId: targetColId };

      if (targetIndex === -1) {
        return [...listWithoutDragged, updatedCard];
      }

      const insertIndex = position === 'above' ? targetIndex : targetIndex + 1;
      const result = [...listWithoutDragged];
      result.splice(insertIndex, 0, updatedCard);
      return result;
    });

    this.onDragEnd();
  }

  private triggerSnapBackNotification(msg: string): void {
    this.invalidMoveWarning.set(msg);
    if (this.warningTimeout) {
      clearTimeout(this.warningTimeout);
    }
    this.warningTimeout = setTimeout(() => {
      this.invalidMoveWarning.set(null);
    }, 3200);
  }
}
```
