```ts file=src/app/app.ts
import { Component, OnDestroy, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { NgIconComponent, provideIcons } from '@ng-icons/core';
import { lucideApple, lucidePlay, lucidePause } from '@ng-icons/lucide';

export type PomodoroMode = 'focus' | 'shortPause' | 'longPause';

export interface TimerConfig {
  focus: number;       // tempo em segundos. Padrão 1500 (25 minutos)
  shortPause: number;  // tempo em segundos. Padrão 300 (5 minutos)
  longPause: number;   // tempo em segundos. Padrão 900 (15 minutos)
}

export interface PomodoroState {
  currentMode: PomodoroMode;
  timeRemainingInSeconds: number;
  isRunning: boolean;
  config: TimerConfig;
}

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, FormsModule, NgIconComponent],
  viewProviders: [
    provideIcons({
      lucideApple,
      lucidePlay,
      lucidePause
    })
  ],
  templateUrl: './app.html'
})
export class App implements OnInit, OnDestroy {
  public state: PomodoroState = {
    currentMode: 'focus',
    timeRemainingInSeconds: 1500,
    isRunning: false,
    config: {
      focus: 1500,
      shortPause: 300,
      longPause: 900
    }
  };

  private timerIntervalId: ReturnType<typeof setInterval> | null = null;
  private beepAudioDataUri = '';

  public readonly modeDetails: Record<PomodoroMode, { title: string; color: string; bgClass: string; borderClass: string; textClass: string }> = {
    focus: {
      title: 'Foco Total',
      color: '#D74747',
      bgClass: 'bg-[#D74747]',
      borderClass: 'border-[#D74747]',
      textClass: 'text-[#D74747]'
    },
    shortPause: {
      title: 'Pausa Curta',
      color: '#658B5C',
      bgClass: 'bg-[#658B5C]',
      borderClass: 'border-[#658B5C]',
      textClass: 'text-[#658B5C]'
    },
    longPause: {
      title: 'Pausa Longa',
      color: '#5684D9',
      bgClass: 'bg-[#5684D9]',
      borderClass: 'border-[#5684D9]',
      textClass: 'text-[#5684D9]'
    }
  };

  ngOnInit(): void {
    this.beepAudioDataUri = this.generateBeepDataUri();
  }

  ngOnDestroy(): void {
    this.stopTimer();
  }

  get currentBorderColor(): string {
    return this.modeDetails[this.state.currentMode].borderClass;
  }

  get currentModeTitle(): string {
    return this.modeDetails[this.state.currentMode].title;
  }

  get formattedTime(): string {
    const mins = Math.floor(this.state.timeRemainingInSeconds / 60);
    const secs = this.state.timeRemainingInSeconds % 60;
    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  }

  get focusMinutes(): number {
    return Math.floor(this.state.config.focus / 60);
  }

  get shortPauseMinutes(): number {
    return Math.floor(this.state.config.shortPause / 60);
  }

  get longPauseMinutes(): number {
    return Math.floor(this.state.config.longPause / 60);
  }

  public toggleTimer(): void {
    if (this.state.isRunning) {
      this.pauseTimer();
    } else {
      this.startTimer();
    }
  }

  public startTimer(): void {
    if (this.state.timeRemainingInSeconds <= 0) {
      this.state.timeRemainingInSeconds = this.state.config[this.state.currentMode];
    }

    this.state.isRunning = true;

    if (this.timerIntervalId) {
      clearInterval(this.timerIntervalId);
    }

    this.timerIntervalId = setInterval(() => {
      if (this.state.timeRemainingInSeconds > 0) {
        this.state.timeRemainingInSeconds--;

        if (this.state.timeRemainingInSeconds === 0) {
          this.pauseTimer();
          this.playAlertSound();
        }
      } else {
        this.pauseTimer();
      }
    }, 1000);
  }

  public pauseTimer(): void {
    this.state.isRunning = false;
    this.stopTimer();
  }

  private stopTimer(): void {
    if (this.timerIntervalId) {
      clearInterval(this.timerIntervalId);
      this.timerIntervalId = null;
    }
  }

  public setMode(mode: PomodoroMode): void {
    if (this.state.currentMode === mode) {
      return;
    }

    this.pauseTimer();
    this.state.currentMode = mode;
    this.state.timeRemainingInSeconds = this.state.config[mode];
  }

  public onMinutesConfigChange(mode: PomodoroMode, event: Event): void {
    const inputElement = event.target as HTMLInputElement;
    const minutesValue = Number(inputElement.value);

    if (!isNaN(minutesValue) && minutesValue > 0) {
      const newSeconds = Math.floor(minutesValue * 60);
      this.state.config[mode] = newSeconds;

      if (this.state.currentMode === mode && !this.state.isRunning) {
        this.state.timeRemainingInSeconds = newSeconds;
      }
    }
  }

  private playAlertSound(): void {
    try {
      const audio = new Audio(this.beepAudioDataUri);
      const playPromise = audio.play();
      if (playPromise !== undefined) {
        playPromise.catch(() => {
          // Fallback silencioso se o navegador bloquear autoplay
        });
      }
    } catch {
      // Fallback
    }
  }

  private generateBeepDataUri(): string {
    const sampleRate = 8000;
    const duration = 0.5;
    const numSamples = Math.floor(sampleRate * duration);
    const buffer = new ArrayBuffer(44 + numSamples);
    const view = new DataView(buffer);

    // RIFF
    view.setUint32(0, 0x52494646, false);
    view.setUint32(4, 36 + numSamples, true);
    view.setUint32(8, 0x57415645, false);

    // fmt
    view.setUint32(12, 0x666d7420, false);
    view.setUint32(16, 16, true);
    view.setUint16(20, 1, true);
    view.setUint16(22, 1, true);
    view.setUint32(24, sampleRate, true);
    view.setUint32(28, sampleRate, true);
    view.setUint16(32, 1, true);
    view.setUint16(34, 8, true);

    // data
    view.setUint32(36, 0x64617461, false);
    view.setUint32(40, numSamples, true);

    const freq = 880;
    for (let i = 0; i < numSamples; i++) {
      const t = i / sampleRate;
      const sample = Math.sin(2 * Math.PI * freq * t);
      view.setUint8(44 + i, Math.floor((sample + 1) * 127.5));
    }

    let binary = '';
    const bytes = new Uint8Array(buffer);
    for (let i = 0; i < bytes.length; i++) {
      binary += String.fromCharCode(bytes[i]);
    }

    return 'data:audio/wav;base64,' + btoa(binary);
  }
}
```
