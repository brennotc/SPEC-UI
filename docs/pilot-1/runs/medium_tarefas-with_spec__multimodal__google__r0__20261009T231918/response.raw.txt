```ts file=src/app/models/task.model.ts
export type PriorityLevel = 'URGENTE' | 'MÉDIO' | 'RELAXA';

export interface Task {
  id: string;
  description: string;
  dueDate: string; // Formato YYYY-MM-DD
  priority: PriorityLevel;
  completed: boolean;
}

export interface TaskGroup {
  groupTitle: string; // "Hoje", "Amanhã" ou a data formatada
  dateKey: string;
  tasks: Task[];
}

export interface TaskFilter {
  date?: string;
  priority?: PriorityLevel | 'TODAS';
}
```

```ts file=src/app/app.ts
import { Component, signal, computed } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { NgIconComponent, provideIcons } from '@ng-icons/core';
import {
  lucideListTodo,
  lucidePencil,
  lucideTrash2,
  lucidePlus,
  lucideCheck,
  lucideX
} from '@ng-icons/lucide';
import { Task, PriorityLevel, TaskGroup } from './models/task.model';

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, FormsModule, NgIconComponent],
  providers: [
    provideIcons({
      lucideListTodo,
      lucidePencil,
      lucideTrash2,
      lucidePlus,
      lucideCheck,
      lucideX
    })
  ],
  templateUrl: './app.html'
})
export class App {
  // Cores de referência
  readonly COLOR_HEADER = '#60C2E5';
  readonly COLOR_URGENTE = '#FF0000';
  readonly COLOR_MEDIO = '#8D8A31';
  readonly COLOR_RELAXA = '#0F633F';
  readonly COLOR_CINZA = '#D5D0D0';

  // Gerador de datas dinâmicas de referência para manter "Hoje" e "Amanhã" sempre sincronizados
  private todayIso = this.formatDateToIso(new Date());
  private tomorrowIso = this.formatDateToIso(
    new Date(Date.now() + 24 * 60 * 60 * 1000)
  );

  // Lista reativa de tarefas
  tasks = signal<Task[]>([
    {
      id: '1',
      description: 'Lavar a louça',
      dueDate: this.todayIso,
      priority: 'URGENTE',
      completed: false
    },
    {
      id: '2',
      description: 'Organizar o lixo para a reciclagem',
      dueDate: this.todayIso,
      priority: 'RELAXA',
      completed: false
    },
    {
      id: '3',
      description: 'Fazer mercado',
      dueDate: this.todayIso,
      priority: 'MÉDIO',
      completed: false
    },
    {
      id: '4',
      dueDate: this.tomorrowIso,
      description: 'Lavar roupas',
      priority: 'URGENTE',
      completed: false
    },
    {
      id: '5',
      dueDate: this.tomorrowIso,
      description: 'Organizar o lixo para a reciclagem',
      priority: 'RELAXA',
      completed: false
    }
  ]);

  // Filtros de busca
  filterDateMask = signal<string>(''); // formato DD/MM/AAAA
  filterPriority = signal<string>('Selecione'); // "Selecione" ou 'URGENTE' | 'MÉDIO' | 'RELAXA'

  // Formulário Nova Tarefa
  newTaskDate = signal<string>(this.isoToBrazilian(this.todayIso));
  newTaskPriority = signal<string>('');
  newTaskDescription = signal<string>('');

  // Modo de Edição em Linha
  editingTaskId = signal<string | null>(null);
  editDate = signal<string>('');
  editPriority = signal<PriorityLevel>('MÉDIO');
  editDescription = signal<string>('');

  // Mapeamento de peso para ordenar prioridades: URGENTE (1) > MÉDIO (2) > RELAXA (3)
  private readonly priorityWeight: Record<PriorityLevel, number> = {
    URGENTE: 1,
    MÉDIO: 2,
    RELAXA: 3
  };

  // Tarefas filtradas e agrupadas por dia
  taskGroups = computed<TaskGroup[]>(() => {
    const all = this.tasks();
    const filterP = this.filterPriority();
    const filterD = this.filterDateMask().trim();

    // Filtrar por prioridade
    let filtered = all.filter((t) => {
      if (filterP && filterP !== 'Selecione' && filterP !== 'TODAS') {
        if (t.priority !== filterP) return false;
      }
      return true;
    });

    // Filtrar por data se preenchida (DD/MM/AAAA completa ou parcial)
    if (filterD.replace(/[^0-9]/g, '').length === 8) {
      const filterIso = this.brazilianToIso(filterD);
      filtered = filtered.filter((t) => t.dueDate === filterIso);
    } else if (filterD.length > 0 && filterD !== '__/__/____') {
      filtered = filtered.filter((t) => {
        const brDate = this.isoToBrazilian(t.dueDate);
        return brDate.startsWith(filterD);
      });
    }

    // Agrupar por data (dueDate)
    const groupsMap = new Map<string, Task[]>();
    for (const task of filtered) {
      if (!groupsMap.has(task.dueDate)) {
        groupsMap.set(task.dueDate, []);
      }
      groupsMap.get(task.dueDate)!.push(task);
    }

    // Ordenar grupos cronologicamente
    const sortedDateKeys = Array.from(groupsMap.keys()).sort();

    return sortedDateKeys.map((dateKey) => {
      // Ordenar tarefas dentro do grupo por prioridade
      const tasksInGroup = groupsMap.get(dateKey)!.sort((a, b) => {
        return (
          this.priorityWeight[a.priority] - this.priorityWeight[b.priority]
        );
      });

      return {
        dateKey,
        groupTitle: this.resolveGroupTitle(dateKey),
        tasks: tasksInGroup
      };
    });
  });

  // Métodos de formatação e datas
  private formatDateToIso(d: Date): string {
    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
  }

  private isoToBrazilian(isoDate: string): string {
    if (!isoDate || !isoDate.includes('-')) return '';
    const [y, m, d] = isoDate.split('-');
    return `${d}/${m}/${y}`;
  }

  private brazilianToIso(brDate: string): string {
    const cleaned = brDate.replace(/[^0-9]/g, '');
    if (cleaned.length !== 8) return '';
    const day = cleaned.substring(0, 2);
    const month = cleaned.substring(2, 4);
    const year = cleaned.substring(4, 8);
    return `${year}-${month}-${day}`;
  }

  private resolveGroupTitle(isoDate: string): string {
    if (isoDate === this.todayIso) return 'Hoje';
    if (isoDate === this.tomorrowIso) return 'Amanhã';
    return this.isoToBrazilian(isoDate);
  }

  // Máscara interativa de data (DD/MM/AAAA)
  onDateInput(event: Event, targetSignal: 'filter' | 'new' | 'edit'): void {
    const input = event.target as HTMLInputElement;
    let value = input.value.replace(/\D/g, '');

    if (value.length > 8) {
      value = value.substring(0, 8);
    }

    let masked = value;
    if (value.length > 2) {
      masked = value.substring(0, 2) + '/' + value.substring(2);
    }
    if (value.length > 4) {
      masked =
        value.substring(0, 2) +
        '/' +
        value.substring(2, 4) +
        '/' +
        value.substring(4, 8);
    }

    input.value = masked;

    if (targetSignal === 'filter') {
      this.filterDateMask.set(masked);
    } else if (targetSignal === 'new') {
      this.newTaskDate.set(masked);
    } else if (targetSignal === 'edit') {
      this.editDate.set(masked);
    }
  }

  // Ações das tarefas
  toggleTask(taskId: string): void {
    this.tasks.update((list) =>
      list.map((t) => (t.id === taskId ? { ...t, completed: !t.completed } : t))
    );
  }

  deleteTask(taskId: string): void {
    this.tasks.update((list) => list.filter((t) => t.id !== taskId));
    if (this.editingTaskId() === taskId) {
      this.editingTaskId.set(null);
    }
  }

  startEditing(task: Task): void {
    this.editingTaskId.set(task.id);
    this.editDate.set(this.isoToBrazilian(task.dueDate));
    this.editPriority.set(task.priority);
    this.editDescription.set(task.description);
  }

  cancelEditing(): void {
    this.editingTaskId.set(null);
  }

  saveEditing(taskId: string): void {
    const desc = this.editDescription().trim();
    if (!desc) return;

    let isoDate = this.brazilianToIso(this.editDate());
    if (!isoDate) {
      isoDate = this.todayIso;
    }

    this.tasks.update((list) =>
      list.map((t) =>
        t.id === taskId
          ? {
              ...t,
              description: desc,
              dueDate: isoDate,
              priority: this.editPriority()
            }
          : t
      )
    );

    this.editingTaskId.set(null);
  }

  // Adição de nova tarefa via rodapé
  addTask(): void {
    const desc = this.newTaskDescription().trim();
    if (!desc) return;

    let isoDate = this.brazilianToIso(this.newTaskDate());
    if (!isoDate) {
      isoDate = this.todayIso;
    }

    const priorityInput = this.newTaskPriority();
    const priority: PriorityLevel =
      priorityInput === 'URGENTE' ||
      priorityInput === 'MÉDIO' ||
      priorityInput === 'RELAXA'
        ? priorityInput
        : 'MÉDIO';

    const newTask: Task = {
      id: Date.now().toString(),
      description: desc,
      dueDate: isoDate,
      priority,
      completed: false
    };

    this.tasks.update((list) => [...list, newTask]);

    // Limpar campos
    this.newTaskDescription.set('');
    this.newTaskPriority.set('');
    this.newTaskDate.set('');
  }
}
```
