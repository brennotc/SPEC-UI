```ts file=src/app/types.ts
export type PriorityLevel = 'URGENTE' | 'MÉDIO' | 'RELAXA';

export interface Task {
  id: string;
  description: string;
  dueDate: string; // Formato YYYY-MM-DD
  priority: PriorityLevel;
  completed: boolean;
}

export interface TaskGroup {
  groupTitle: string; // "Hoje", "Amanhã" ou a data formatada
  dateKey: string;
  tasks: Task[];
}

export interface TaskFilter {
  date?: string;
  priority?: PriorityLevel | 'TODAS';
}

export const PRIORITY_WEIGHT: Record<PriorityLevel, number> = {
  URGENTE: 3,
  'MÉDIO': 2,
  RELAXA: 1
};
```

```ts file=src/app/app.ts
import { Component, computed, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { NgIcon, provideIcons } from '@ng-icons/core';
import { heroListBullet } from '@ng-icons/heroicons/outline';
import {
  lucidePencil,
  lucideTrash2,
  lucideCheck,
  lucidePlus,
  lucideChevronDown,
  lucideX
} from '@ng-icons/lucide';
import { PriorityLevel, Task, TaskGroup, PRIORITY_WEIGHT } from './types';

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, NgIcon],
  viewProviders: [
    provideIcons({
      heroListBullet,
      lucidePencil,
      lucideTrash2,
      lucideCheck,
      lucidePlus,
      lucideChevronDown,
      lucideX
    })
  ],
  templateUrl: './app.html',
  styleUrls: ['./app.css']
})
export class App {
  // Helpers para cálculo de datas padrão
  private todayStr = this.getTodayIso();
  private tomorrowStr = this.getOffsetIso(1);
  private dayAfterStr = this.getOffsetIso(2);

  // Estado das Tarefas
  readonly tasks = signal<Task[]>([
    {
      id: 'task-1',
      description: 'Reunião de alinhamento com equipe de produto',
      dueDate: this.todayStr,
      priority: 'URGENTE',
      completed: false
    },
    {
      id: 'task-2',
      description: 'Revisar PR de autenticação e permissões',
      dueDate: this.todayStr,
      priority: 'MÉDIO',
      completed: false
    },
    {
      id: 'task-3',
      description: 'Organizar pastas de downloads e backup',
      dueDate: this.todayStr,
      priority: 'RELAXA',
      completed: true
    },
    {
      id: 'task-4',
      description: 'Preparar apresentação de arquitetura Angular 20',
      dueDate: this.tomorrowStr,
      priority: 'URGENTE',
      completed: false
    },
    {
      id: 'task-5',
      description: 'Atualizar documentação das APIs no Confluence',
      dueDate: this.tomorrowStr,
      priority: 'MÉDIO',
      completed: false
    },
    {
      id: 'task-6',
      description: 'Pesquisar novas extensões do Tailwind CSS',
      dueDate: this.dayAfterStr,
      priority: 'RELAXA',
      completed: false
    }
  ]);

  // Filtros
  readonly filterDateDisplay = signal<string>(''); // formato DD/MM/AAAA
  readonly filterPriority = signal<PriorityLevel | 'TODAS'>('TODAS');

  // Formulário de Adição
  readonly newTaskDate = signal<string>(this.isoToDisplay(this.todayStr));
  readonly newTaskPriority = signal<PriorityLevel>('URGENTE');
  readonly newTaskDescription = signal<string>('');

  // Modo de Edição
  readonly editingTaskId = signal<string | null>(null);
  readonly editDateInput = signal<string>('');
  readonly editPriorityInput = signal<PriorityLevel>('URGENTE');
  readonly editDescriptionInput = signal<string>('');

  // Tarefas agrupadas e ordenadas por prioridade
  readonly groupedTasks = computed<TaskGroup[]>(() => {
    const all = this.tasks();
    const filterD = this.filterDateDisplay().trim();
    const filterP = this.filterPriority();

    let filtered = all;

    // Filtro por data
    if (filterD.length === 10) {
      const isoFilter = this.displayToIso(filterD);
      filtered = filtered.filter((t) => t.dueDate === isoFilter);
    }

    // Filtro por prioridade
    if (filterP !== 'TODAS') {
      filtered = filtered.filter((t) => t.priority === filterP);
    }

    // Agrupar por data (dueDate)
    const groupsMap = new Map<string, Task[]>();
    for (const task of filtered) {
      if (!groupsMap.has(task.dueDate)) {
        groupsMap.set(task.dueDate, []);
      }
      groupsMap.get(task.dueDate)!.push(task);
    }

    // Ordenar as datas cronologicamente
    const sortedDateKeys = Array.from(groupsMap.keys()).sort();

    // Ordenar as tarefas dentro de cada grupo da mais prioritária para a menos prioritária
    return sortedDateKeys.map((dateKey) => {
      const tasksInGroup = groupsMap.get(dateKey) || [];
      const sortedTasks = [...tasksInGroup].sort((a, b) => {
        const weightDiff = PRIORITY_WEIGHT[b.priority] - PRIORITY_WEIGHT[a.priority];
        if (weightDiff !== 0) return weightDiff;
        return a.description.localeCompare(b.description);
      });

      return {
        groupTitle: this.resolveGroupTitle(dateKey),
        dateKey,
        tasks: sortedTasks
      };
    });
  });

  // Utilitários de data
  private getTodayIso(): string {
    const d = new Date();
    return this.formatIsoDate(d);
  }

  private getOffsetIso(offsetDays: number): string {
    const d = new Date();
    d.setDate(d.getDate() + offsetDays);
    return this.formatIsoDate(d);
  }

  private formatIsoDate(d: Date): string {
    const y = d.getFullYear();
    const m = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${y}-${m}-${day}`;
  }

  private isoToDisplay(iso: string): string {
    if (!iso || !iso.includes('-')) return '';
    const [y, m, d] = iso.split('-');
    if (!y || !m || !d) return '';
    return `${d.padStart(2, '0')}/${m.padStart(2, '0')}/${y}`;
  }

  private displayToIso(display: string): string {
    const parts = display.split('/');
    if (parts.length !== 3) return '';
    const [d, m, y] = parts;
    return `${y}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`;
  }

  private applyDateMask(value: string): string {
    const digits = value.replace(/\D/g, '').slice(0, 8);
    if (digits.length <= 2) return digits;
    if (digits.length <= 4) return `${digits.slice(0, 2)}/${digits.slice(2)}`;
    return `${digits.slice(0, 2)}/${digits.slice(2, 4)}/${digits.slice(4)}`;
  }

  resolveGroupTitle(dateKey: string): string {
    const today = this.getTodayIso();
    const tomorrow = this.getOffsetIso(1);

    if (dateKey === today) {
      return 'Hoje';
    }
    if (dateKey === tomorrow) {
      return 'Amanhã';
    }
    return this.isoToDisplay(dateKey);
  }

  getPriorityColor(priority: PriorityLevel): string {
    switch (priority) {
      case 'URGENTE':
        return '#FF0000';
      case 'MÉDIO':
        return '#8D8A31';
      case 'RELAXA':
        return '#0F633F';
      default:
        return '#D5D0D0';
    }
  }

  // Ações de Filtro
  onFilterDateInput(event: Event): void {
    const input = event.target as HTMLInputElement;
    const masked = this.applyDateMask(input.value);
    input.value = masked;
    this.filterDateDisplay.set(masked);
  }

  onFilterPriorityChange(event: Event): void {
    const select = event.target as HTMLSelectElement;
    this.filterPriority.set(select.value as PriorityLevel | 'TODAS');
  }

  clearFilters(): void {
    this.filterDateDisplay.set('');
    this.filterPriority.set('TODAS');
  }

  // Ações de Tarefa
  toggleTaskCompletion(id: string): void {
    this.tasks.update((list) =>
      list.map((task) =>
        task.id === id ? { ...task, completed: !task.completed } : task
      )
    );
  }

  deleteTask(id: string): void {
    this.tasks.update((list) => list.filter((task) => task.id !== id));
    if (this.editingTaskId() === id) {
      this.cancelEditing();
    }
  }

  startEditingTask(task: Task): void {
    this.editingTaskId.set(task.id);
    this.editDateInput.set(this.isoToDisplay(task.dueDate));
    this.editPriorityInput.set(task.priority);
    this.editDescriptionInput.set(task.description);
  }

  onEditDateChange(event: Event): void {
    const input = event.target as HTMLInputElement;
    const masked = this.applyDateMask(input.value);
    input.value = masked;
    this.editDateInput.set(masked);
  }

  onEditPriorityChange(event: Event): void {
    const select = event.target as HTMLSelectElement;
    this.editPriorityInput.set(select.value as PriorityLevel);
  }

  onEditDescChange(event: Event): void {
    const input = event.target as HTMLInputElement;
    this.editDescriptionInput.set(input.value);
  }

  saveEditingTask(): void {
    const id = this.editingTaskId();
    if (!id) return;

    let isoDate = this.displayToIso(this.editDateInput());
    if (!isoDate || isoDate.length !== 10) {
      isoDate = this.getTodayIso();
    }

    const desc = this.editDescriptionInput().trim();
    if (!desc) return;

    const priority = this.editPriorityInput();

    this.tasks.update((list) =>
      list.map((task) =>
        task.id === id
          ? {
              ...task,
              dueDate: isoDate,
              priority,
              description: desc
            }
          : task
      )
    );

    this.cancelEditing();
  }

  cancelEditing(): void {
    this.editingTaskId.set(null);
    this.editDateInput.set('');
    this.editDescriptionInput.set('');
  }

  // Ações de Adição
  onNewTaskDateInput(event: Event): void {
    const input = event.target as HTMLInputElement;
    const masked = this.applyDateMask(input.value);
    input.value = masked;
    this.newTaskDate.set(masked);
  }

  onNewTaskPriorityChange(event: Event): void {
    const select = event.target as HTMLSelectElement;
    this.newTaskPriority.set(select.value as PriorityLevel);
  }

  onSubmitNewTask(event: Event): void {
    event.preventDefault();
    const desc = this.newTaskDescription().trim();
    if (!desc) return;

    let iso = this.displayToIso(this.newTaskDate());
    if (!iso || iso.length !== 10) {
      iso = this.getTodayIso();
    }

    const newTask: Task = {
      id: 'task-' + Date.now() + '-' + Math.random().toString(36).substring(2, 6),
      description: desc,
      dueDate: iso,
      priority: this.newTaskPriority(),
      completed: false
    };

    this.tasks.update((list) => [...list, newTask]);

    // Limpar os inputs do formulário
    this.newTaskDescription.set('');
    this.newTaskPriority.set('URGENTE');
    this.newTaskDate.set(this.isoToDisplay(this.getTodayIso()));
  }
}
```
