export class App implements OnDestroy {
  // Configurações visuais por modo
  readonly modeConfig: Record<PomodoroMode, ModeConfig> = {
    focus: {
      name: 'Foco Total',
      defaultMinutes: 25,
      colorBorder: 'border-red-600',
      colorText: 'text-red-600',
      colorBadgeBg: 'bg-red-50 border-red-200',
      colorBadgeText: 'text-red-700',
      colorBtn: 'bg-red-600 hover:bg-red-700 focus:ring-red-300',
      ringProgressColor: '#dc2626'
    },
    shortBreak: {
      name: 'Pausa Curta',
      defaultMinutes: 5,
      colorBorder: 'border-emerald-600',
      colorText: 'text-emerald-600',
      colorBadgeBg: 'bg-emerald-50 border-emerald-200',
      colorBadgeText: 'text-emerald-700',
      colorBtn: 'bg-emerald-600 hover:bg-emerald-700 focus:ring-emerald-300',
      ringProgressColor: '#059669'
    },
    longBreak: {
      name: 'Pausa Longa',
      defaultMinutes: 15,
      colorBorder: 'border-blue-600',
      colorText: 'text-blue-600',
      colorBadgeBg: 'bg-blue-50 border-blue-200',
      colorBadgeText: 'text-blue-700',
      colorBtn: 'bg-blue-600 hover:bg-blue-700 focus:ring-blue-300',
      ringProgressColor: '#2563eb'
    }
  };

  // Raio do círculo SVG: r=46%, perímetro normalizado 2 * PI * 130 ~ 816
  readonly circumference = 2 * Math.PI * 135;

  // Estados com Signals
  readonly currentMode = signal<PomodoroMode>('focus');
  readonly focusMinutes = signal<number>(25);
  readonly shortBreakMinutes = signal<number>(5);
  readonly longBreakMinutes = signal<number>(15);

  readonly timeLeft = signal<number>(25 * 60);
  readonly totalDuration = signal<number>(25 * 60);
  readonly isRunning = signal<boolean>(false);
  readonly soundEnabled = signal<boolean>(true);

  private timerInterval: any = null;
  private audioCtx: AudioContext | null = null;

  // Formatação do tempo: 00:00
  readonly formattedTime = computed(() => {
    const seconds = this.timeLeft();
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    const pad = (n: number) => n.toString().padStart(2, '0');
    return `${pad(mins)}:${pad(secs)}`;
  });

  // Cálculo da barra de progresso circular
  readonly progressOffset = computed(() => {
    const total = this.totalDuration();
    if (total <= 0) return 0;
    const current = this.timeLeft();
    const progress = (total - current) / total;
    return this.circumference * (1 - progress);
  });

  constructor() {
    // Sincroniza o título da aba com o tempo restante
    effect(() => {
      const time = this.formattedTime();
      const modeName = this.modeConfig[this.currentMode()].name;
      if (typeof document !== 'undefined') {
        document.title = `${time} - ${modeName} | Pomodoro`;
      }
    });
  }

  setMode(mode: PomodoroMode): void {
    if (this.currentMode() === mode) return;

    this.pauseTimer();
    this.currentMode.set(mode);
    const mins = this.getMinutesForMode(mode);
    this.totalDuration.set(mins * 60);
    this.timeLeft.set(mins * 60);
  }

  toggleTimer(): void {
    if (this.isRunning()) {
      this.pauseTimer();
    } else {
      if (this.timeLeft() <= 0) {
        // Se já estava zerado, reinicia
        const mins = this.getMinutesForMode(this.currentMode());
        this.timeLeft.set(mins * 60);
      }
      this.startTimer();
    }
  }

  startTimer(): void {
    this.isRunning.set(true);
    this.clearInterval();

    this.timerInterval = setInterval(() => {
      const current = this.timeLeft();
      if (current > 1) {
        this.timeLeft.set(current - 1);
      } else if (current === 1) {
        this.timeLeft.set(0);
        this.onTimerCompleted();
      }
    }, 1000);
  }

  pauseTimer(): void {
    this.isRunning.set(false);
    this.clearInterval();
  }

  resetCurrentTimer(): void {
    this.pauseTimer();
    const mins = this.getMinutesForMode(this.currentMode());
    this.totalDuration.set(mins * 60);
    this.timeLeft.set(mins * 60);
  }

  onMinutesChange(mode: PomodoroMode, rawValue: any): void {
    let value = parseInt(rawValue, 10);
    if (isNaN(value) || value < 1) {
      value = 1;
    } else if (value > 180) {
      value = 180;
    }

    if (mode === 'focus') this.focusMinutes.set(value);
    if (mode === 'shortBreak') this.shortBreakMinutes.set(value);
    if (mode === 'longBreak') this.longBreakMinutes.set(value);

    // Se o modo editado for o modo atual e não estiver rodando, atualiza o tempo restante
    if (this.currentMode() === mode && !this.isRunning()) {
      this.totalDuration.set(value * 60);
      this.timeLeft.set(value * 60);
    }
  }

  resetDefaults(): void {
    this.pauseTimer();
    this.focusMinutes.set(25);
    this.shortBreakMinutes.set(5);
    this.longBreakMinutes.set(15);
    const mins = this.getMinutesForMode(this.currentMode());
    this.totalDuration.set(mins * 60);
    this.timeLeft.set(mins * 60);
  }

  private getMinutesForMode(mode: PomodoroMode): number {
    switch (mode) {
      case 'focus': return this.focusMinutes();
      case 'shortBreak': return this.shortBreakMinutes();
      case 'longBreak': return this.longBreakMinutes();
    }
  }

  private onTimerCompleted(): void {
    this.pauseTimer();
    this.playChime();
  }

  /**
   * Som de aviso agradável e harmônico construído puramente com Web Audio API.
   * Não precisa de arquivos externos de áudio e funciona offline.
   */
  private playChime(): void {
    if (!this.soundEnabled()) return;

    try {
      const AudioCtxClass = window.AudioContext || (window as any).webkitAudioContext;
      if (!AudioCtxClass) return;

      if (!this.audioCtx || this.audioCtx.state === 'closed') {
        this.audioCtx = new AudioCtxClass();
      }

      if (this.audioCtx.state === 'suspended') {
        this.audioCtx.resume();
      }

      const now = this.audioCtx.currentTime;

      // Acorde suave de sino (E5, G#5, B5, E6)
      const notes = [659.25, 830.61, 987.77, 1318.51];

      notes.forEach((freq, idx) => {
        if (!this.audioCtx) return;
        const osc = this.audioCtx.createOscillator();
        const gain = this.audioCtx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(freq, now + idx * 0.12);

        gain.gain.setValueAtTime(0, now + idx * 0.12);
        gain.gain.linearRampToValueAtTime(0.25, now + idx * 0.12 + 0.04);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + idx * 0.12 + 1.2);

        osc.connect(gain);
        gain.connect(this.audioCtx.destination);

        osc.start(now + idx * 0.12);
        osc.stop(now + idx * 0.12 + 1.25);
      });
    } catch (e) {
      console.warn('Erro ao reproduzir áudio com Web Audio API:', e);
    }
  }

  private clearInterval(): void {
    if (this.timerInterval) {
      clearInterval(this.timerInterval);
      this.timerInterval = null;
    }
  }

  ngOnDestroy(): void {
    this.clearInterval();
    if (this.audioCtx && this.audioCtx.state !== 'closed') {
      this.audioCtx.close().catch(() => {});
    }
  }
}
