```ts file=src/app/models/task.model.ts
export type PriorityLevel = 'URGENTE' | 'MÉDIO' | 'RELAXA';

export type TaskGroupType = 'HOJE' | 'AMANHA' | 'OUTROS';

export interface TaskItem {
  id: string;
  title: string;
  date: string; // formato DD/MM/YYYY
  priority: PriorityLevel;
  completed: boolean;
}
```

```ts file=src/app/app.ts
import { Component, signal, computed } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { NgIconComponent, provideIcons } from '@ng-icons/core';
import {
  lucideListChecks,
  lucidePencil,
  lucideTrash2,
  lucidePlus,
  lucideCheck,
  lucideX,
  lucideCalendar,
  lucideSlidersHorizontal
} from '@ng-icons/lucide';
import { PriorityLevel, TaskItem } from './models/task.model';

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, FormsModule, NgIconComponent],
  viewProviders: [
    provideIcons({
      lucideListChecks,
      lucidePencil,
      lucideTrash2,
      lucidePlus,
      lucideCheck,
      lucideX,
      lucideCalendar,
      lucideSlidersHorizontal
    })
  ],
  templateUrl: './app.html',
  styleUrls: ['./app.css']
})
export class App {
  // Filtros
  filterDate = signal<string>('');
  filterPriority = signal<string>('');

  // Formulário Nova Tarefa
  newTaskDate = signal<string>('');
  newTaskPriority = signal<PriorityLevel | ''>('');
  newTaskTitle = signal<string>('');

  // Edição inline
  editingTaskId = signal<string | null>(null);
  editingTaskTitle = signal<string>('');
  editingTaskPriority = signal<PriorityLevel>('MÉDIO');
  editingTaskDate = signal<string>('');

  // Helper de datas atuais no formato DD/MM/YYYY
  readonly todayStr: string;
  readonly tomorrowStr: string;

  tasks = signal<TaskItem[]>([]);

  constructor() {
    const now = new Date();
    const tomorrow = new Date();
    tomorrow.setDate(now.getDate() + 1);

    this.todayStr = this.formatDateObj(now);
    this.tomorrowStr = this.formatDateObj(tomorrow);

    // Inicializa com as tarefas que aparecem na imagem de exemplo
    this.tasks.set([
      {
        id: '1',
        title: 'Lavar a louça',
        date: this.todayStr,
        priority: 'URGENTE',
        completed: false
      },
      {
        id: '2',
        title: 'Organizar o lixo para a reciclagem',
        date: this.todayStr,
        priority: 'RELAXA',
        completed: false
      },
      {
        id: '3',
        title: 'Fazer mercado',
        date: this.todayStr,
        priority: 'MÉDIO',
        completed: false
      },
      {
        id: '4',
        title: 'Lavar roupas',
        date: this.tomorrowStr,
        priority: 'URGENTE',
        completed: false
      },
      {
        id: '5',
        title: 'Organizar o lixo para a reciclagem',
        date: this.tomorrowStr,
        priority: 'RELAXA',
        completed: false
      }
    ]);
  }

  // Grupos filtrados
  filteredGroups = computed(() => {
    const fDate = this.filterDate().trim();
    const fPrio = this.filterPriority().trim();

    const filtered = this.tasks().filter((task) => {
      const matchDate = !fDate || task.date.includes(fDate);
      const matchPrio = !fPrio || task.priority === fPrio;
      return matchDate && matchPrio;
    });

    // Separar em Hoje, Amanhã e Outros (agrupados por data)
    const hojeTasks = filtered.filter((t) => t.date === this.todayStr);
    const amanhaTasks = filtered.filter((t) => t.date === this.tomorrowStr);
    const outrosTasks = filtered.filter(
      (t) => t.date !== this.todayStr && t.date !== this.tomorrowStr
    );

    // Agrupar 'outros' por data ordenada
    const groups: { title: string; date: string; tasks: TaskItem[] }[] = [];

    if (hojeTasks.length > 0 || (!fDate && !fPrio)) {
      groups.push({ title: 'Hoje', date: this.todayStr, tasks: hojeTasks });
    }

    if (amanhaTasks.length > 0 || (!fDate && !fPrio)) {
      groups.push({ title: 'Amanhã', date: this.tomorrowStr, tasks: amanhaTasks });
    }

    // Organizar demais datas distintas
    const distinctDates = Array.from(new Set(outrosTasks.map((t) => t.date))).sort();
    for (const d of distinctDates) {
      groups.push({
        title: this.getGroupDisplayTitle(d),
        date: d,
        tasks: outrosTasks.filter((t) => t.date === d)
      });
    }

    return groups;
  });

  private formatDateObj(d: Date): string {
    const day = String(d.getDate()).padStart(2, '0');
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const year = d.getFullYear();
    return `${day}/${month}/${year}`;
  }

  private getGroupDisplayTitle(dateStr: string): string {
    const parts = dateStr.split('/');
    if (parts.length === 3) {
      const dateObj = new Date(Number(parts[2]), Number(parts[1]) - 1, Number(parts[0]));
      if (!isNaN(dateObj.getTime())) {
        const diaSemana = dateObj.toLocaleDateString('pt-BR', { weekday: 'long' });
        const capitalized = diaSemana.charAt(0).toUpperCase() + diaSemana.slice(1);
        return `${capitalized} (${dateStr})`;
      }
    }
    return dateStr;
  }

  // Máscara para inputs de data (DD/MM/AAAA)
  applyDateMask(event: Event, targetSignal: 'filter' | 'new' | 'edit'): void {
    const input = event.target as HTMLInputElement;
    let digits = input.value.replace(/\D/g, '');
    if (digits.length > 8) digits = digits.slice(0, 8);

    let masked = '';
    if (digits.length > 4) {
      masked = `${digits.slice(0, 2)}/${digits.slice(2, 4)}/${digits.slice(4)}`;
    } else if (digits.length > 2) {
      masked = `${digits.slice(0, 2)}/${digits.slice(2)}`;
    } else {
      masked = digits;
    }

    input.value = masked;
    if (targetSignal === 'filter') {
      this.filterDate.set(masked);
    } else if (targetSignal === 'new') {
      this.newTaskDate.set(masked);
    } else if (targetSignal === 'edit') {
      this.editingTaskDate.set(masked);
    }
  }

  // Preenchimento de atalho para Hoje no input novo se estiver vazio
  setDefaultDateIfEmpty(): void {
    if (!this.newTaskDate()) {
      this.newTaskDate.set(this.todayStr);
    }
  }

  toggleTask(taskId: string): void {
    this.tasks.update((items) =>
      items.map((t) => (t.id === taskId ? { ...t, completed: !t.completed } : t))
    );
  }

  deleteTask(taskId: string): void {
    this.tasks.update((items) => items.filter((t) => t.id !== taskId));
    if (this.editingTaskId() === taskId) {
      this.cancelEditing();
    }
  }

  startEditing(task: TaskItem): void {
    this.editingTaskId.set(task.id);
    this.editingTaskTitle.set(task.title);
    this.editingTaskPriority.set(task.priority);
    this.editingTaskDate.set(task.date);
  }

  cancelEditing(): void {
    this.editingTaskId.set(null);
    this.editingTaskTitle.set('');
    this.editingTaskDate.set('');
  }

  saveEditing(): void {
    const id = this.editingTaskId();
    const title = this.editingTaskTitle().trim();
    if (!id || !title) return;

    this.tasks.update((items) =>
      items.map((t) =>
        t.id === id
          ? {
              ...t,
              title,
              priority: this.editingTaskPriority(),
              date: this.editingTaskDate() || t.date
            }
          : t
      )
    );
    this.cancelEditing();
  }

  addNewTask(): void {
    const title = this.newTaskTitle().trim();
    const priority = this.newTaskPriority();
    const dateInput = this.newTaskDate().trim();

    if (!title) {
      alert('Por favor, informe a descrição da tarefa.');
      return;
    }

    const taskDate = dateInput || this.todayStr;
    const taskPriority = (priority as PriorityLevel) || 'MÉDIO';

    const newTask: TaskItem = {
      id: Date.now().toString(),
      title,
      date: taskDate,
      priority: taskPriority,
      completed: false
    };

    this.tasks.update((items) => [...items, newTask]);

    // Limpar campos
    this.newTaskTitle.set('');
    this.newTaskPriority.set('');
    this.newTaskDate.set('');
  }

  getBadgeClass(priority: PriorityLevel): string {
    switch (priority) {
      case 'URGENTE':
        return 'bg-[#ff0000] text-white';
      case 'MÉDIO':
        return 'bg-[#968725] text-white';
      case 'RELAXA':
        return 'bg-[#0f5b35] text-white';
      default:
        return 'bg-zinc-700 text-white';
    }
  }
}
```
