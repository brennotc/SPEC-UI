```ts file=src/app/app.ts
import { Component, OnDestroy, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { NgIconComponent, provideIcons } from '@ng-icons/core';
import { lucidePlay, lucidePause, lucideApple } from '@ng-icons/lucide';

export type PomodoroMode = 'focus' | 'shortPause' | 'longPause';

export interface TimerConfig {
  focus: number;       // em segundos (padrão 1500 = 25 min)
  shortPause: number;  // em segundos (padrão 300 = 5 min)
  longPause: number;   // em segundos (padrão 900 = 15 min)
}

export interface PomodoroState {
  currentMode: PomodoroMode;
  timeRemainingInSeconds: number;
  isRunning: boolean;
  config: TimerConfig;
}

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, FormsModule, NgIconComponent],
  providers: [
    provideIcons({
      lucidePlay,
      lucidePause,
      lucideApple
    })
  ],
  templateUrl: './app.html',
  styleUrls: ['./app.css']
})
export class App implements OnInit, OnDestroy {
  readonly modeColors: Record<PomodoroMode, string> = {
    focus: '#D74747',
    shortPause: '#658B5C',
    longPause: '#5684D9'
  };

  readonly modeLabels: Record<PomodoroMode, string> = {
    focus: 'Foco Total',
    shortPause: 'Pausa Curta',
    longPause: 'Pausa Longa'
  };

  state: PomodoroState = {
    currentMode: 'focus',
    timeRemainingInSeconds: 1500,
    isRunning: false,
    config: {
      focus: 1500,
      shortPause: 300,
      longPause: 900
    }
  };

  // Minutos para os inputs numéricos
  minutesInput: Record<PomodoroMode, number> = {
    focus: 25,
    shortPause: 5,
    longPause: 15
  };

  private timerInterval: ReturnType<typeof setInterval> | null = null;
  private alarmAudio: HTMLAudioElement | null = null;

  ngOnInit(): void {
    this.initAudio();
  }

  ngOnDestroy(): void {
    this.stopTimer();
  }

  get currentColor(): string {
    return this.modeColors[this.state.currentMode];
  }

  get currentLabel(): string {
    return this.modeLabels[this.state.currentMode];
  }

  get formattedTime(): string {
    return this.formatTime(this.state.timeRemainingInSeconds);
  }

  formatTime(totalSeconds: number): string {
    const minutes = Math.floor(Math.max(0, totalSeconds) / 60);
    const seconds = Math.floor(Math.max(0, totalSeconds) % 60);
    const mm = minutes.toString().padStart(2, '0');
    const ss = seconds.toString().padStart(2, '0');
    return `${mm}:${ss}`;
  }

  setMode(mode: PomodoroMode): void {
    if (this.state.currentMode === mode && this.state.isRunning) {
      return;
    }
    this.stopTimer();
    this.state.currentMode = mode;
    this.state.timeRemainingInSeconds = this.state.config[mode];
  }

  toggleTimer(): void {
    if (this.state.isRunning) {
      this.pauseTimer();
    } else {
      this.startTimer();
    }
  }

  startTimer(): void {
    if (this.state.timeRemainingInSeconds <= 0) {
      this.state.timeRemainingInSeconds = this.state.config[this.state.currentMode];
    }
    this.state.isRunning = true;
    this.timerInterval = setInterval(() => {
      if (this.state.timeRemainingInSeconds > 0) {
        this.state.timeRemainingInSeconds--;
        if (this.state.timeRemainingInSeconds === 0) {
          this.handleTimerFinished();
        }
      }
    }, 1000);
  }

  pauseTimer(): void {
    this.stopTimer();
  }

  private stopTimer(): void {
    this.state.isRunning = false;
    if (this.timerInterval) {
      clearInterval(this.timerInterval);
      this.timerInterval = null;
    }
  }

  private handleTimerFinished(): void {
    this.stopTimer();
    this.playAlarm();
  }

  private initAudio(): void {
    try {
      // Criar um áudio curto via base64 WAV para funcionar 100% offline
      const wavBase64 = 'data:audio/wav;base64,UklGRnoGAABXQVZFZm10IBAAAAABAAEAQB8AAEAfAAABAAgAZGF0YV4GAACBhYqFbF1fdJivrJBhNjVgodDbq2EcHCek3fLbuWscHCGW1/Teu2scHiid2vLbu2sdHi2h2/LbvWwhIS6i3PPcvnAiJC2k3PPdv3IlJS+m3vTewXMnJjCn3vTgwnYqKTOo3/ThxHksLTap4PXiynwwLjmr4ffky38zLz2u4vjmzYI2MD2w4/rnz4U4MT6y5Pvo0Yg6Mz+05fzo04w9NEK35/3p1ZBE';
      this.alarmAudio = new Audio(wavBase64);
      this.alarmAudio.volume = 1.0;
    } catch {
      this.alarmAudio = new Audio();
    }
  }

  private playAlarm(): void {
    try {
      if (!this.alarmAudio) {
        this.alarmAudio = new Audio();
      }
      this.alarmAudio.currentTime = 0;
      const playPromise = this.alarmAudio.play();
      if (playPromise !== undefined) {
        playPromise.catch(() => {
          // Fallback para Web Audio API se o navegador bloquear autoplay
          this.playBeepWebAudio();
        });
      }
    } catch {
      this.playBeepWebAudio();
    }
  }

  private playBeepWebAudio(): void {
    try {
      const audioCtx = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(880, audioCtx.currentTime);
      gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 1.2);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + 1.2);
    } catch {
      // Falha silenciosa
    }
  }

  onMinutesChange(mode: PomodoroMode, newMinutes: number): void {
    if (this.state.isRunning) return;

    const validatedMinutes = Math.max(1, Number(newMinutes) || 1);
    this.minutesInput[mode] = validatedMinutes;
    this.state.config[mode] = validatedMinutes * 60;

    // Se o modo alterado for o modo atual, atualiza o tempo restante principal imediatamente
    if (this.state.currentMode === mode) {
      this.state.timeRemainingInSeconds = this.state.config[mode];
    }
  }
}
```
