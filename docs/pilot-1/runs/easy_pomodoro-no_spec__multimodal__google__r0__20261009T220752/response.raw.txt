```ts file=src/app/app.ts
import { Component, OnDestroy, OnInit, signal, computed, effect } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';

type PomodoroMode = 'focus' | 'shortBreak' | 'longBreak';

interface ModeConfig {
  id: PomodoroMode;
  name: string;
  defaultMinutes: number;
  colorHex: string;
  bgHex: string;
  badgeBg: string;
}

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './app.html',
  styleUrls: ['./app.css']
})
export class App implements OnInit, OnDestroy {
  // Configurações dos modos
  readonly modes: Record<PomodoroMode, ModeConfig> = {
    focus: {
      id: 'focus',
      name: 'Foco Total',
      defaultMinutes: 25,
      colorHex: '#df5353',
      bgHex: '#df5353',
      badgeBg: 'bg-[#d84a4a]'
    },
    shortBreak: {
      id: 'shortBreak',
      name: 'Pausa Curta',
      defaultMinutes: 5,
      colorHex: '#4d8055',
      bgHex: '#4d8055',
      badgeBg: 'bg-[#477e52]'
    },
    longBreak: {
      id: 'longBreak',
      name: 'Pausa Longa',
      defaultMinutes: 15,
      colorHex: '#487ecb',
      bgHex: '#487ecb',
      badgeBg: 'bg-[#437bc6]'
    }
  };

  // Estado atual
  currentMode = signal<PomodoroMode>('focus');
  isRunning = signal<boolean>(false);
  timeRemaining = signal<number>(25 * 60);

  // Valores de tempo configurados em minutos para cada modo
  focusMinutes = signal<number>(25);
  shortBreakMinutes = signal<number>(5);
  longBreakMinutes = signal<number>(15);

  // Strings para edição no input
  focusInput = signal<string>('25:00');
  shortBreakInput = signal<string>('05:00');
  longBreakInput = signal<string>('15:00');

  private timerInterval: any = null;
  private audioCtx: AudioContext | null = null;

  currentModeConfig = computed(() => this.modes[this.currentMode()]);

  formattedTime = computed(() => {
    const totalSeconds = this.timeRemaining();
    const minutes = Math.floor(totalSeconds / 60);
    const seconds = totalSeconds % 60;
    return `${this.padZero(minutes)}:${this.padZero(seconds)}`;
  });

  ngOnInit(): void {
    this.updateTitle();
  }

  ngOnDestroy(): void {
    this.stopTimer();
    if (this.audioCtx) {
      this.audioCtx.close().catch(() => {});
    }
  }

  private padZero(num: number): string {
    return num.toString().padStart(2, '0');
  }

  togglePlayPause(): void {
    if (this.isRunning()) {
      this.pauseTimer();
    } else {
      this.startTimer();
    }
  }

  startTimer(): void {
    if (this.isRunning()) return;

    this.initAudio();
    this.isRunning.set(true);

    this.timerInterval = setInterval(() => {
      const current = this.timeRemaining();
      if (current > 1) {
        this.timeRemaining.set(current - 1);
        this.updateTitle();
      } else {
        this.timeRemaining.set(0);
        this.updateTitle();
        this.onTimerFinished();
      }
    }, 1000);
  }

  pauseTimer(): void {
    this.isRunning.set(false);
    this.stopTimer();
  }

  private stopTimer(): void {
    if (this.timerInterval) {
      clearInterval(this.timerInterval);
      this.timerInterval = null;
    }
  }

  setMode(mode: PomodoroMode): void {
    if (this.currentMode() === mode) return;

    this.pauseTimer();
    this.currentMode.set(mode);

    let targetMinutes = 25;
    if (mode === 'focus') targetMinutes = this.focusMinutes();
    else if (mode === 'shortBreak') targetMinutes = this.shortBreakMinutes();
    else if (mode === 'longBreak') targetMinutes = this.longBreakMinutes();

    this.timeRemaining.set(targetMinutes * 60);
    this.updateTitle();
  }

  private onTimerFinished(): void {
    this.pauseTimer();
    this.playBeepSound();

    // Notificação ou transição para o próximo modo
    setTimeout(() => {
      if (this.currentMode() === 'focus') {
        this.setMode('shortBreak');
      } else {
        this.setMode('focus');
      }
    }, 1200);
  }

  // Manipulação dos inputs editáveis
  onInputChange(mode: PomodoroMode, event: Event): void {
    const input = event.target as HTMLInputElement;
    let val = input.value.trim();

    // Parse minutos:segundos ou apenas minutos
    let minutes = 0;
    let seconds = 0;

    if (val.includes(':')) {
      const parts = val.split(':');
      minutes = parseInt(parts[0], 10) || 0;
      seconds = parseInt(parts[1], 10) || 0;
    } else {
      minutes = parseInt(val, 10) || 0;
      seconds = 0;
    }

    // Limites de segurança
    if (minutes < 1 && seconds < 5) {
      minutes = 1;
      seconds = 0;
    }
    if (minutes > 120) {
      minutes = 120;
      seconds = 0;
    }

    const totalSec = minutes * 60 + seconds;
    const formatted = `${this.padZero(minutes)}:${this.padZero(seconds)}`;

    if (mode === 'focus') {
      this.focusMinutes.set(minutes);
      this.focusInput.set(formatted);
      if (this.currentMode() === 'focus' && !this.isRunning()) {
        this.timeRemaining.set(totalSec);
      }
    } else if (mode === 'shortBreak') {
      this.shortBreakMinutes.set(minutes);
      this.shortBreakInput.set(formatted);
      if (this.currentMode() === 'shortBreak' && !this.isRunning()) {
        this.timeRemaining.set(totalSec);
      }
    } else if (mode === 'longBreak') {
      this.longBreakMinutes.set(minutes);
      this.longBreakInput.set(formatted);
      if (this.currentMode() === 'longBreak' && !this.isRunning()) {
        this.timeRemaining.set(totalSec);
      }
    }
    this.updateTitle();
  }

  private updateTitle(): void {
    const status = this.isRunning() ? '▶ ' : '';
    document.title = `${status}${this.formattedTime()} - Pomodoro`;
  }

  private initAudio(): void {
    if (!this.audioCtx) {
      const AudioContextClass = window.AudioContext || (window as any).webkitAudioContext;
      if (AudioContextClass) {
        this.audioCtx = new AudioContextClass();
      }
    }
    if (this.audioCtx && this.audioCtx.state === 'suspended') {
      this.audioCtx.resume();
    }
  }

  // Alerta sonoro agradável usando Web Audio API sem dependências externas
  playBeepSound(): void {
    try {
      this.initAudio();
      if (!this.audioCtx) return;

      const now = this.audioCtx.currentTime;

      // Tríade suave de sinos/bips (880Hz, 1174Hz, 1760Hz)
      const freqs = [880, 1174.66, 1760];
      freqs.forEach((freq, idx) => {
        const osc = this.audioCtx!.createOscillator();
        const gain = this.audioCtx!.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(freq, now + idx * 0.16);

        gain.gain.setValueAtTime(0.001, now + idx * 0.16);
        gain.gain.exponentialRampToValueAtTime(0.3, now + idx * 0.16 + 0.03);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + idx * 0.16 + 0.45);

        osc.connect(gain);
        gain.connect(this.audioCtx!.destination);

        osc.start(now + idx * 0.16);
        osc.stop(now + idx * 0.16 + 0.5);
      });
    } catch {
      // Ignora erro caso navegador bloqueie som sem interação
    }
  }
}
```
